Usage: helios opstack [OPTIONS] --network <NETWORK>

Options:
  -n, --network <NETWORK>
          
  -b, --rpc-bind-ip <RPC_BIND_IP>
          [env: RPC_BIND_IP=] [default: 127.0.0.1]
  -p, --rpc-port <RPC_PORT>
          [env: RPC_PORT=] [default: 8545]
  -e, --execution-rpc <EXECUTION_RPC>
          [env: EXECUTION_RPC=]
      --tui
          Enable terminal user interface
      --execution-verifiable-api <EXECUTION_VERIFIABLE_API>
          [env: EXECUTION_VERIFIABLE_API=]
  -c, --consensus-rpc <CONSENSUS_RPC>
          [env: CONSENSUS_RPC=]
  -w, --ethereum-checkpoint <CHECKPOINT>
          Set custom weak subjectivity checkpoint for chosen Ethereum network. Helios uses this to sync and trustlessly fetch the correct unsafe signer address used by <NETWORK> [env: ETHEREUM_CHECKPOINT=]
  -l, --ethereum-load-external-fallback
          Enable fallback for weak subjectivity checkpoint. Use if --ethereum-checkpoint fails. [env: ETHEREUM_LOAD_EXTERNAL_FALLBACK=]
  -h, --help
          Print help
