def count_vowels(text):
  """
  Counts the number of vowels (a, e, i, o, u, case-insensitive) in a given string.

  Args:
    text: The string to analyze.

  Returns:
    The total count of vowels in the string.
  """
  vowels = "aeiouAEIOU"
  count = 0
  for char in text:
    if char in vowels:
      count += 1
  return count
